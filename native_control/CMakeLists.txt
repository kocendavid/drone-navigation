cmake_minimum_required(VERSION 3.20)
project(isaac_control LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(px4_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_library(isaac_control SHARED control.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/control.cpp" API_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/time_core.hpp" CORE_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_HASH)
string(SHA256 SOURCE_HASH "${API_HASH}${CORE_HASH}${CMAKE_HASH}")
target_compile_definitions(isaac_control PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN IC_SOURCE_HASH="${SOURCE_HASH}")
set_target_properties(isaac_control PROPERTIES OUTPUT_NAME "isaac_control_${SOURCE_HASH}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/current_build.json"
  CONTENT "{\"library\":\"$<TARGET_FILE_NAME:isaac_control>\",\"source_hash\":\"${SOURCE_HASH}\"}\n")
ament_target_dependencies(isaac_control rclcpp px4_msgs)
target_link_libraries(isaac_control nlohmann_json::nlohmann_json)
if(WIN32)
  target_link_libraries(isaac_control ws2_32)
endif()
if(MSVC)
  target_compile_options(isaac_control PRIVATE /W4 /EHsc)
else()
  target_compile_options(isaac_control PRIVATE -Wall -Wextra -Wpedantic)
endif()
