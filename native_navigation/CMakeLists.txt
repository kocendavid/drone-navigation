cmake_minimum_required(VERSION 3.20)
project(isaac_navigation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(isaac_navigation SHARED kernels.cpp)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/kernels.cpp" KERNEL_HASH)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" CMAKE_HASH)
string(SHA256 SOURCE_HASH "${KERNEL_HASH}${CMAKE_HASH}")
target_compile_definitions(isaac_navigation PRIVATE NAV_SOURCE_HASH="${SOURCE_HASH}")
set_target_properties(isaac_navigation PROPERTIES OUTPUT_NAME "isaac_navigation_${SOURCE_HASH}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/current_build.json"
  CONTENT "{\"library\":\"$<TARGET_FILE_NAME:isaac_navigation>\",\"source_hash\":\"${SOURCE_HASH}\"}\n")
if(MSVC)
  target_compile_options(isaac_navigation PRIVATE /W4 /EHsc)
else()
  target_compile_options(isaac_navigation PRIVATE -Wall -Wextra -Wpedantic)
endif()
